# Runtime dependencies, pinned to the set this project is developed and tested
# against. Generated from a working venv, not guessed — see docs/installation.md.
#
#   pip install -r requirements.txt
#
# numpy is held at 1.x on purpose: the opencv / onnxruntime / torch wheels are
# built against the 1.x ABI and segfault or refuse to import against 2.x.
numpy<2

# 4.10 ships FFMPEG:YES, which RTSP needs. Do not float this to 5.x without
# checking `cv2.getBuildInformation()` for FFMPEG support first.
#
# CAVEAT: ultralytics depends on `opencv-python` — the GUI build — so pip will
# install that too, over the top of this one. Both unpack into the same `cv2/`
# directory, so this pin does not actually win; check with `python -c "import
# cv2; print(cv2.__version__)"`. It has been benign so far (the GUI build also
# ships FFMPEG:YES), but it is a collision, not a plan. To get a single clean
# headless install:
#
#   pip uninstall -y opencv-python opencv-python-headless
#   pip install "opencv-python-headless==4.10.*"
#
# The Dockerfile does exactly that, and asserts the result.
opencv-python-headless==4.10.*

PyYAML>=6.0
psutil>=5.9          # system-health pane (CPU / memory / storage)

# Detection + tracking (stage B). Pulls torch — the big download.
ultralytics>=8.3

# ByteTrack's assignment solver. Ultralytics imports it lazily and only fails at
# the first tracked frame, so pin it here rather than discover it at runtime.
lap>=0.5

# Semantic search (search plan Phase 6) — `pip install sentence-transformers`,
# then `python -m intelligence_os.semantic` to build the index. ~120MB of
# weights, downloaded once and run entirely locally: no keys, no network at
# query time. Without it, search falls back to the lexical index — narrower
# recall, identical answers, and the suite stays green either way (prove it with
# INTELLIGENCE_OS_NO_SEMANTIC=1).
#
# The upper bounds are not caution, they are the installed torch. transformers
# 5.x requires torch>=2.4 and ultralytics brings 2.2 here, so an unpinned
# install resolves to a transformers that imports and then fails at the first
# model load. Raise both together, after checking `torch.__version__`.
# sentence-transformers>=3,<4
# transformers>=4.41,<5
